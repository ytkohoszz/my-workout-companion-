function sanitizeDraftPlanList(raw: unknown): PlannedExerciseItem[] {
  if (!Array.isArray(raw)) return [];
  const out: PlannedExerciseItem[] = [];
  for (const item of raw) {
    if (!item || typeof item !== "object") continue;
    const e = item as Record<string, unknown>;
    const name = typeof e.name === "string" ? e.name.trim() : "";
    if (!name) continue;
    out.push({
      id:
        typeof e.id === "string" && e.id.trim()
          ? e.id.trim()
          : makePlanItemId(),
      name,
      weight:
        typeof e.weight === "number" && Number.isFinite(e.weight) && e.weight >= 0
          ? e.weight
          : 0,
      sets:
        typeof e.sets === "number" && Number.isFinite(e.sets) && e.sets >= 1
          ? Math.round(e.sets)
          : 4,
      target:
        typeof e.target === "string" && e.target.trim()
          ? e.target.trim()
          : "MAX",
      rest:
        typeof e.rest === "number" && Number.isFinite(e.rest) && e.rest >= 5
          ? Math.round(e.rest)
          : 150,
    });
  }
  return out;
}

function migrateSettings(raw: unknown): WorkoutPluginSettings {
  const s =
    raw && typeof raw === "object"
      ? (raw as Partial<WorkoutPluginSettings>)
      : {};

  const timerRaw =
    s.timer && typeof s.timer === "object"
      ? (s.timer as Partial<ActiveRestTimerState>)
      : {};
  const timer: ActiveRestTimerState = {
    sessionPath:
      typeof timerRaw.sessionPath === "string" ? timerRaw.sessionPath : "",
    startAt: typeof timerRaw.startAt === "number" ? timerRaw.startAt : 0,
    endAt: typeof timerRaw.endAt === "number" ? timerRaw.endAt : 0,
    durationSec:
      typeof timerRaw.durationSec === "number" && timerRaw.durationSec > 0
        ? timerRaw.durationSec
        : 120,
    notifiedEndAt:
      typeof timerRaw.notifiedEndAt === "number" ? timerRaw.notifiedEndAt : 0,
  };

  const draftBySession: Record<string, SessionDraftState> = {};
  if (s.draftBySession && typeof s.draftBySession === "object") {
    for (const [k, v] of Object.entries(
      s.draftBySession as Record<string, unknown>
    )) {
      if (v && typeof v === "object") {
        const d = v as Partial<SessionDraftState> & { selectedExKey?: string };
        const fallbackName =
          typeof d.selectedExName === "string" && d.selectedExName
            ? d.selectedExName
            : typeof d.selectedExKey === "string"
              ? d.selectedExKey.split("__")[0] ?? "Подтягивания"
              : "Подтягивания";
        draftBySession[k] = {
          selectedPlanItemId:
            typeof d.selectedPlanItemId === "string"
              ? d.selectedPlanItemId
              : "",
          selectedExName: fallbackName,
          customWeight:
            typeof d.customWeight === "number" ? d.customWeight : 0,
          draftReps: typeof d.draftReps === "number" ? d.draftReps : 10,
          restDurationSec:
            typeof d.restDurationSec === "number" && d.restDurationSec > 0
              ? d.restDurationSec
              : 150,
          plannedExercises: sanitizeDraftPlanList(d.plannedExercises),
        };
      }
    }
  }

  return {
    workoutsFolder:
      typeof s.workoutsFolder === "string" && s.workoutsFolder.trim()
        ? s.workoutsFolder.trim()
        : DEFAULT_SETTINGS.workoutsFolder,
    diaryFolder:
      typeof s.diaryFolder === "string" && s.diaryFolder.trim()
        ? s.diaryFolder.trim()
        : DEFAULT_SETTINGS.diaryFolder,
    soundEnabled:
      typeof s.soundEnabled === "boolean"
        ? s.soundEnabled
        : DEFAULT_SETTINGS.soundEnabled,
    timer,
    draftBySession,
  };
}

export default class MyWorkoutCompanionPlugin extends Plugin {
  settings!: WorkoutPluginSettings;
  programStore!: ProgramStore;
  timer!: WorkoutRestTimerService;
  private saveTimer: number | null = null;

  async onload() {
    const rawData = await this.loadData();
    this.settings = migrateSettings(rawData);

    const pluginDir =
      this.manifest.dir ||
      `${this.app.vault.configDir}/plugins/${this.manifest.id}`;
    this.programStore = new ProgramStore(this.app, pluginDir);
    await this.programStore.load(rawData);

    this.timer = new WorkoutRestTimerService(this);
    this.addSettingTab(new WorkoutCompanionSettingTab(this.app, this));

    this.addRibbonIcon("flame", "Открыть терминал тренировки", () => {
      this.openWorkout();
    });

    this.registerObsidianProtocolHandler("my-workout", (params) => {
      this.openWorkout({
        sessionPath: params.session || undefined,
      });
    });

    this.addCommand({
      id: "open-workout-companion",
      name: "Открыть терминал тренировки",
      callback: () => this.openWorkout(),
    });

    this.registerDomEvent(document, "visibilitychange", () => {
      if (document.visibilityState === "visible") {
        this.timer.checkAndSyncNow();
      }
    });
    this.registerDomEvent(window, "focus", () => {
      this.timer.checkAndSyncNow();
    });

    this.app.workspace.onLayoutReady(() => {
      this.timer.checkAndSyncNow();
    });
  }

  openWorkout(options: OpenWorkoutOptions = {}) {
    new WorkoutCompanionModal(this.app, this, options).open();
  }

  requestSaveSettings() {
    if (this.saveTimer !== null) window.clearTimeout(this.saveTimer);
    this.saveTimer = window.setTimeout(() => {
      this.saveTimer = null;
      void this.saveData(this.settings);
    }, 250);
  }

  onunload() {
    if (this.saveTimer !== null) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
    }
    this.programStore?.flush();
    this.timer?.destroy();
  }
}

    if (preFileDraft && this.activeSession) {
      this.plugin.settings.draftBySession[key] = preFileDraft;
      delete this.plugin.settings.draftBySession[`date:${todayISO}`];
      return preFileDraft;

      this.activeSession && this.activeSession.loggedSets.length > 0

      return;

    const finishedAt = this.plugin.timer.getLastFinishedBannerEndAt();

      attr: { type: "button" },
    });
    customExToggleBtn.onclick = () => {
      this.isNewExerciseFormOpen = !this.isNewExerciseFormOpen;
      void this.refreshAll();
    };

    // Инлайн-форма создания своего упражнения
    if (this.isNewExerciseFormOpen) {
      const customBox = builderCard.createDiv("my-wc-inline-drawer");
      const exNameInp = customBox.createEl("input", {
        cls: "my-wc-set-target",
        attr: {
          type: "text",
          placeholder: "Название упражнения…",
          value: this.newExNameDraft,
        },
      });
      exNameInp.oninput = () => {
        this.newExNameDraft = exNameInp.value;
      };

      const kindSel = customBox.createEl("select", { cls: "dropdown" });
      kindSel.createEl("option", {
        value: "повторы",
        text: "Повторения (повт.)",
      });
      const secOpt = kindSel.createEl("option", {
        value: "секунды",
        text: "Время (секунды, с)",
      });
      if (this.newExFieldDraft === "секунды") secOpt.selected = true;
      kindSel.onchange = () => {
        this.newExFieldDraft =
          kindSel.value === "секунды" ? "секунды" : "повторы";
      };

      const saveExBtn = customBox.createEl("button", {
        cls: "mod-cta my-wc-btn",
        text: "Создать и добавить",
        attr: { type: "button" },
      });
      saveExBtn.onclick = () => {
        const created = this.plugin.programStore.addCustomExercise(
          this.newExNameDraft,
          this.newExFieldDraft
        );
        if (!created) {
          new Notice("Введи название упражнения.");
          return;
        }
        this.newExNameDraft = "";
        this.isNewExerciseFormOpen = false;
        this.addExerciseToPlan(draft, created);
        new Notice(`Упражнение «${created.name}» добавлено.`);
      };
    }

    // Список добавленных в тренировку упражнений с явными подписями над каждым полем
    if (draft.plannedExercises.length === 0) {
      builderCard.createDiv({
        cls: "my-wc-sets-empty",
        text: "Список пока пуст — выбери упражнения кнопками выше.",
      });
    } else {
      const listEl = builderCard.createDiv("my-wc-plan-cards-list");
      const catalog = this.plugin.programStore.getCatalog();

      draft.plannedExercises.forEach((item, idx) => {
        const itemCard = listEl.createDiv("my-wc-plan-item-card");

        // Верхняя строка карточки упражнения: номер + выбор упражнения + кнопки порядка и удаления
        const topBar = itemCard.createDiv("my-wc-plan-item-top");
        topBar.createSpan({
          cls: "my-wc-plan-item-num",
          text: `${idx + 1}.`,
        });

        const exSel = topBar.createEl("select", {
          cls: "dropdown my-wc-plan-item-select",
        });
        const hasInCat = catalog.some((c) => c.name === item.name);
        if (!hasInCat) {
          exSel.createEl("option", { value: item.name, text: item.name });
        }
        for (const c of catalog) {
          const o = exSel.createEl("option", { value: c.name, text: c.name });
          if (c.name === item.name) o.selected = true;
        }
        exSel.onchange = () => {
          item.name = exSel.value;
          if (draft.selectedPlanItemId === item.id) {
            draft.selectedExName = item.name;
          }
          this.plugin.requestSaveSettings();
          void this.refreshAll();
        };

        const rowTools = topBar.createDiv("my-wc-plan-item-tools");
        const upBtn = rowTools.createEl("button", {
          cls: "my-wc-btn",
          text: "↑",
          attr: { type: "button", title: "Переместить выше" },
        });
        upBtn.disabled = idx === 0;
        upBtn.onclick = () => {
          if (idx <= 0) return;
          const prev = draft.plannedExercises[idx - 1]!;
          draft.plannedExercises[idx - 1] = item;
          draft.plannedExercises[idx] = prev;
          this.plugin.requestSaveSettings();
          void this.refreshAll();
        };

        const downBtn = rowTools.createEl("button", {
          cls: "my-wc-btn",
          text: "↓",
          attr: { type: "button", title: "Переместить ниже" },
        });
        downBtn.disabled = idx === draft.plannedExercises.length - 1;
        downBtn.onclick = () => {
          if (idx >= draft.plannedExercises.length - 1) return;
          const next = draft.plannedExercises[idx + 1]!;
          draft.plannedExercises[idx + 1] = item;
          draft.plannedExercises[idx] = next;
          this.plugin.requestSaveSettings();
          void this.refreshAll();
        };

        const delBtn = rowTools.createEl("button", {
          cls: "my-wc-btn",
          text: "✕ Убрать",
          attr: { type: "button", title: "Убрать из сегодняшней тренировки" },
        });
        delBtn.onclick = () => {
          draft.plannedExercises.splice(idx, 1);
          if (draft.selectedPlanItemId === item.id) {
            const fallback = draft.plannedExercises[0];
            draft.selectedPlanItemId = fallback?.id ?? "";
            if (fallback) {
              draft.selectedExName = fallback.name;
              draft.customWeight = fallback.weight;
              draft.restDurationSec = fallback.rest;
            }
          }
          this.plugin.requestSaveSettings();
          void this.refreshAll();
        };

        // Нижняя строка карточки: 4 подписанных поля (Доп. вес · Подходов · Цель · Отдых)
        const fieldsGrid = itemCard.createDiv("my-wc-plan-fields-grid");

        // Поле 1: Доп. вес (кг)
        const fWeight = fieldsGrid.createEl("label", { cls: "my-wc-labeled-field" });
        fWeight.createSpan({ cls: "my-wc-field-caption", text: "Доп. вес (кг)" });
        const wInp = fWeight.createEl("input", {
          cls: "my-wc-field-input",
          attr: {
            type: "number",
            inputmode: "decimal",
            step: "0.5",
            min: "0",
            value: String(item.weight),
          },
        });
        wInp.oninput = () => {
          const n = Number(wInp.value);
          if (Number.isFinite(n) && n >= 0) {
            item.weight = n;
            if (draft.selectedPlanItemId === item.id) {
              draft.customWeight = n;
            }
            this.plugin.requestSaveSettings();
          }
        };

        // Поле 2: Подходов
        const fSets = fieldsGrid.createEl("label", { cls: "my-wc-labeled-field" });
        fSets.createSpan({ cls: "my-wc-field-caption", text: "Подходов" });
        const setsInp = fSets.createEl("input", {
          cls: "my-wc-field-input",
          attr: {
            type: "number",
            inputmode: "numeric",
            min: "1",
            value: String(item.sets),
          },
        });
        setsInp.oninput = () => {
          const n = Math.round(Number(setsInp.value));
          if (Number.isFinite(n) && n >= 1) {
            item.sets = n;
            this.plugin.requestSaveSettings();
          }
        };

        // Поле 3: Цель повторений
        const fTarget = fieldsGrid.createEl("label", { cls: "my-wc-labeled-field" });
        fTarget.createSpan({ cls: "my-wc-field-caption", text: "Цель (MAX / 8–10)" });
        const targetInp = fTarget.createEl("input", {
          cls: "my-wc-field-input",
          attr: {
            type: "text",
            placeholder: "MAX",
            value: item.target,
          },
        });
        targetInp.oninput = () => {
          item.target = targetInp.value.trim() || "MAX";
          this.plugin.requestSaveSettings();
        };

        // Поле 4: Отдых (сек)
        const fRest = fieldsGrid.createEl("label", { cls: "my-wc-labeled-field" });
        fRest.createSpan({ cls: "my-wc-field-caption", text: "Отдых (сек)" });
        const restInp = fRest.createEl("input", {
          cls: "my-wc-field-input",
          attr: {
            type: "number",
            inputmode: "numeric",
            min: "5",
            step: "15",
            value: String(item.rest),
          },
        });
        restInp.oninput = () => {
          const n = Math.round(Number(restInp.value));
          if (Number.isFinite(n) && n >= 5) {
            item.rest = n;
            if (draft.selectedPlanItemId === item.id) {
              draft.restDurationSec = n;
            }
            this.plugin.requestSaveSettings();
          }
        };
      });
    }
  }

  private renderBody(todayISO: string, draft: SessionDraftState): void {
    this.bodyEl.empty();

    // 1. Блок списка упражнений на сегодня
    this.renderWorkoutBuilderSection(draft);

    // 2. Если файл сессии за сегодня ещё не создан — кнопка старта
    if (!this.activeSession) {
      const startBox = this.bodyEl.createDiv("my-wc-empty-box");
      const startBtn = startBox.createEl("button", {
        cls: "mod-cta my-wc-primary-btn",
        text: "▶ Начать тренировку",
        attr: { type: "button" },
      });
      startBtn.onclick = async () => {
        startBtn.disabled = true;
        this.plugin.timer.unlockAudioOnUserGesture();
        try {
          const title = formatDefaultSessionTitle(
            todayISO,
            draft.plannedExercises
          );
          const file = await createOrOpenWorkoutSessionFile(
            this.app,
            this.plugin.settings,
            todayISO,
            title
          );
          this.plugin.settings.draftBySession[file.path] = draft;
          delete this.plugin.settings.draftBySession[`date:${todayISO}`];
          this.isBuilderExpanded = false;
          this.plugin.requestSaveSettings();
          new Notice(`Тренировка начата: ${title}`);
          await this.refreshAll();
        } catch (err: unknown) {
          new Notice(
            "Ошибка создания сессии: " +
              (err instanceof Error ? err.message : String(err))
          );
          startBtn.disabled = false;
        }
      };
      return;
    }

    const session = this.activeSession;
    const catalog = this.plugin.programStore.getCatalog();
    const activePlanItem =
      draft.plannedExercises.find((p) => p.id === draft.selectedPlanItemId) ??
      null;
    const selectedExName =
      activePlanItem?.name ||
      draft.selectedExName ||
      catalog[0]?.name ||
      "Подтягивания";
    const exMeta = getExerciseMeta(selectedExName, this.plugin.programStore);

    // 3. Карточка записи подхода
    const loggerCard = this.bodyEl.createDiv("my-wc-logger-card");

    const exRow = loggerCard.createDiv("my-wc-ex-row");
    const exSelect = exRow.createEl("select", {
      cls: "dropdown my-wc-ex-select",
    });

    if (draft.plannedExercises.length > 0) {
      const planGroup = exSelect.createEl("optgroup", {
        attr: { label: "Упражнения сегодняшней тренировки" },
      });
      for (const p of draft.plannedExercises) {
        const done = this.countDoneSetsForPlanItem(session, p);
        const wLabel = p.weight > 0 ? `+${p.weight} кг` : "свой вес";
        const o = planGroup.createEl("option", {
          value: `plan:${p.id}`,
          text: `${p.name} (${wLabel} · подход ${done}/${p.sets} · цель ${p.target})`,
        });
        if (activePlanItem && activePlanItem.id === p.id) {
          o.selected = true;
        }
      }
    }

    const catGroup = exSelect.createEl("optgroup", {
      attr: { label: "Все упражнения" },
    });
    for (const c of catalog) {
      const o = catGroup.createEl("option", {
        value: `cat:${c.name}`,
        text: `${c.name} (${c.unit})`,
      });
      if (!activePlanItem && c.name === selectedExName) {
        o.selected = true;
      }
    }

    exSelect.onchange = () => {
      const val = exSelect.value;
      if (val.startsWith("plan:")) {
        const id = val.slice(5);
        const found = draft.plannedExercises.find((p) => p.id === id);
        if (found) {
          draft.selectedPlanItemId = found.id;
          draft.selectedExName = found.name;
          draft.customWeight = found.weight;
          draft.restDurationSec = found.rest;
        }
      } else if (val.startsWith("cat:")) {
        const name = val.slice(4);
        draft.selectedPlanItemId = "";
        draft.selectedExName = name;
      }
      if (this.plugin.settings.timer.endAt <= Date.now()) {
        this.plugin.settings.timer.durationSec = draft.restDurationSec;
      }
      this.plugin.requestSaveSettings();
      void this.refreshAll();
      this.focusSafeContainer();
    };

    if (this.prevRef) {
      const refBar = loggerCard.createDiv("my-wc-prev-ref");
      const wStr =
        this.prevRef.weight > 0 ? `+${this.prevRef.weight} кг` : "свой вес";
      const prText =
        this.prevRef.allTimeBestForExactWeight > 0
          ? ` · рекорд на этом весе: ${this.prevRef.allTimeBestForExactWeight} ${exMeta.unit}`
          : "";
      refBar.setText(
        `В прошлый раз (${formatShortDateRu(this.prevRef.dateISO)}, ${wStr}): ${this.prevRef.sets.join(" · ")} ${exMeta.unit}${prText}`
      );
    }

    const steppersGrid = loggerCard.createDiv("my-wc-steppers-grid");

    // 1. Доп. вес (кг)
    const weightBox = steppersGrid.createDiv("my-wc-stepper-box");
    weightBox.createDiv({ cls: "my-wc-stepper-label", text: "Доп. вес (кг)" });
    const wControls = weightBox.createDiv("my-wc-stepper-controls");

    const wMinusBtn = wControls.createEl("button", {
      cls: "my-wc-step-btn",
      text: "−2",
      attr: { type: "button" },
    });
    const wInput = wControls.createEl("input", {
      cls: "my-wc-step-input",
      attr: {
        type: "number",
        inputmode: "decimal",
        min: "0",
        max: "500",
        step: "0.5",
      },
    });
    wInput.value = String(draft.customWeight);
    const wPlusBtn = wControls.createEl("button", {
      cls: "my-wc-step-btn",
      text: "+2",
      attr: { type: "button" },
    });
    const wZeroBtn = wControls.createEl("button", {
      cls: "my-wc-step-btn is-subtle",
      text: "0",
      attr: { type: "button", title: "Свой вес (0 кг)" },
    });

    const applyWeight = (nextW: number) => {
      const clean = Math.max(0, Math.min(500, Math.round(nextW * 2) / 2));
      draft.customWeight = clean;
      wInput.value = String(clean);
      this.plugin.requestSaveSettings();
    };
    wMinusBtn.onclick = () => applyWeight((Number(wInput.value) || 0) - 2);
    wPlusBtn.onclick = () => applyWeight((Number(wInput.value) || 0) + 2);
    wZeroBtn.onclick = () => applyWeight(0);
    wInput.oninput = () => {
      const n = Number(wInput.value.replace(",", "."));
      if (Number.isFinite(n) && n >= 0) {
        draft.customWeight = n;
        this.plugin.requestSaveSettings();
      }
    };

    // 2. Повторения / Секунды
    const repsBox = steppersGrid.createDiv("my-wc-stepper-box");
    const targetHint = activePlanItem ? ` · цель ${activePlanItem.target}` : "";
    const progressHint = activePlanItem
      ? ` · подход ${Math.min(activePlanItem.sets, this.countDoneSetsForPlanItem(session, activePlanItem) + 1)} из ${activePlanItem.sets}`
      : "";
    repsBox.createDiv({
      cls: "my-wc-stepper-label",
      text:
        (exMeta.field === "секунды" ? "Секунды" : "Повторения") +
        targetHint +
        progressHint,
    });
    const rControls = repsBox.createDiv("my-wc-stepper-controls");

    const rMinusBtn = rControls.createEl("button", {
      cls: "my-wc-step-btn",
      text: "−1",
      attr: { type: "button" },
    });
    const rInput = rControls.createEl("input", {
      cls: "my-wc-step-input",
      attr: {
        type: "number",
        inputmode: "numeric",
        min: "1",
        max: "600",
        step: "1",
      },
    });
    rInput.value = String(draft.draftReps || 10);
    const rPlus1Btn = rControls.createEl("button", {
      cls: "my-wc-step-btn",
      text: "+1",
      attr: { type: "button" },
    });
    const rPlus2Btn = rControls.createEl("button", {
      cls: "my-wc-step-btn",
      text: "+2",
      attr: { type: "button" },
    });

    const applyReps = (nextR: number) => {
      const clean = Math.max(1, Math.min(600, Math.round(nextR)));
      draft.draftReps = clean;
      rInput.value = String(clean);
      this.plugin.requestSaveSettings();
    };
    rMinusBtn.onclick = () => applyReps((Number(rInput.value) || 1) - 1);
    rPlus1Btn.onclick = () => applyReps((Number(rInput.value) || 1) + 1);
    rPlus2Btn.onclick = () => applyReps((Number(rInput.value) || 1) + 2);
    rInput.oninput = () => {
      const n = Math.round(Number(rInput.value));
      if (Number.isFinite(n) && n >= 1) {
        draft.draftReps = n;
        this.plugin.requestSaveSettings();
      }
    };

    const submitBtn = loggerCard.createEl("button", {
      cls: "mod-cta my-wc-primary-btn",
      text: `✓ Записать подход и запустить отдых (${draft.restDurationSec} с)`,
      attr: { type: "button" },
    });

    const doSubmitSet = async () => {
      if (submitBtn.disabled) return;
      this.plugin.timer.unlockAudioOnUserGesture();

      const repsVal = Math.round(Number(rInput.value));
      if (!Number.isFinite(repsVal) || repsVal < 1 || repsVal > 600) {
        new Notice("Укажи значение от 1 до 600.");
        return;
      }
      const weightVal = Math.max(
        0,
        Number(String(wInput.value).replace(",", ".")) || 0
      );

      submitBtn.disabled = true;
      try {
        const prevBestOnSameWeight =
          this.prevRef && Math.abs(this.prevRef.weight - weightVal) < 0.01
            ? this.prevRef.allTimeBestForExactWeight
            : 0;
        const currentSessionBest = session.loggedSets
          .filter(
            (s) =>
              s.name === selectedExName &&
              Math.abs(s.weight - weightVal) < 0.01
          )
          .reduce((m, s) => Math.max(m, s.reps), 0);
        const baselineBest = Math.max(prevBestOnSameWeight, currentSessionBest);
        const isNewRecord = baselineBest > 0 && repsVal > baselineBest;

        const restToStart = draft.restDurationSec || 120;

        await appendSetToWorkoutFile(
          this.app,
          this.plugin.settings,
          this.plugin.programStore,
          session,
          selectedExName,
          repsVal,
          weightVal,
          "Основной"
        );

        draft.customWeight = weightVal;
        draft.draftReps = repsVal;

        let switchedToNextMsg = "";
        if (activePlanItem) {
          const doneNow =
            this.countDoneSetsForPlanItem(session, activePlanItem) + 1;
          if (doneNow >= activePlanItem.sets) {
            const curIdx = draft.plannedExercises.findIndex(
              (p) => p.id === activePlanItem.id
            );
            const nextItem = draft.plannedExercises
              .slice(curIdx + 1)
              .find(
                (p) => this.countDoneSetsForPlanItem(session, p) < p.sets
              );
            if (nextItem) {
              draft.selectedPlanItemId = nextItem.id;
              draft.selectedExName = nextItem.name;
              draft.customWeight = nextItem.weight;
              draft.restDurationSec = nextItem.rest;
              switchedToNextMsg = ` · далее: ${nextItem.name}`;
            }
          }
        }

        this.plugin.timer.start(session.path, restToStart);

        const updatedSession = await hydrateWorkoutSession(this.app, session);
        const lastLogged =
          updatedSession.loggedSets[updatedSession.loggedSets.length - 1];
        this.lastRecordSetLineIndex =
          isNewRecord && lastLogged ? lastLogged.lineIndex : null;

        const recordBadge = isNewRecord ? " 🏆 НОВЫЙ РЕКОРД!" : "";
        new Notice(
          `Записано: ${selectedExName} — ${repsVal} ${exMeta.unit}` +
            (weightVal > 0 ? ` (+${weightVal} кг)` : "") +
            recordBadge +
            switchedToNextMsg
        );
        await this.refreshAll();
      } catch (err: unknown) {
        new Notice(
          "Ошибка записи подхода: " +
            (err instanceof Error ? err.message : String(err))
        );
        submitBtn.disabled = false;
      }
    };

    submitBtn.onclick = () => void doSubmitSet();

    // 4. Выполненные подходы за сегодня
    const setsSection = this.bodyEl.createDiv("my-wc-sets-section");
    const setsHead = setsSection.createDiv("my-wc-sets-head");
    setsHead.createSpan({
      cls: "my-wc-sets-title",
      text: `Выполнено сегодня · ${session.loggedSets.length}`,
    });

    const sessionActions = setsHead.createDiv("my-wc-sets-actions");
    const openNoteBtn = sessionActions.createEl("button", {
      cls: "my-wc-btn",
      text: "📄 Заметка",
      attr: { type: "button", title: "Открыть файл тренировки" },
    });
    openNoteBtn.onclick = () => {
      this.close();
      void this.app.workspace.openLinkText(session.path, "", false);
    };

    const finishBtn = sessionActions.createEl("button", {
      cls: "my-wc-btn",
      text: session.durationMin
        ? `🔄 Обновить время (${session.durationMin} мин)`
        : "✓ Завершить тренировку",
      attr: { type: "button" },
    });
    finishBtn.onclick = async () => {
      finishBtn.disabled = true;
      try {
        const dur = computeSessionDurationMinutes(
          session.startStr,
          session.durationMin,
          session.dateISO || todayISO
        );
        await this.app.fileManager.processFrontMatter(
          session.file,
          (fm: Record<string, unknown>) => {
            fm[WORKOUT_KEYS.durationMin] = dur;
          }
        );
        await syncWorkoutHabitToDiary(
          this.app,
          this.plugin.settings,
          session.dateISO || todayISO
        );
        new Notice(`Тренировка завершена (${dur} мин)`);
        await this.refreshAll();
      } catch (err: unknown) {
        new Notice(
          "Ошибка завершения: " +
            (err instanceof Error ? err.message : String(err))
        );
        finishBtn.disabled = false;
      }
    };

    if (session.loggedSets.length === 0) {
      setsSection.createDiv({
        cls: "my-wc-sets-empty",
        text: "Подходов сегодня пока не записано.",
      });
    } else {
      const totalsByEx = new Map<string, { sum: number; unit: string }>();
      for (const s of session.loggedSets) {
        const m = getExerciseMeta(s.name, this.plugin.programStore);
        const cur = totalsByEx.get(s.name) ?? { sum: 0, unit: m.unit };
        cur.sum += s.reps;
        totalsByEx.set(s.name, cur);
      }
      const volumeSummary = [...totalsByEx.entries()]
        .map(([n, v]) => `${n}: ${v.sum} ${v.unit}`)
        .join(" · ");
      setsSection.createDiv({
        cls: "my-wc-volume-summary",
        text: `Сумма за тренировку: ${volumeSummary}`,
      });

      const pillsList = setsSection.createDiv("my-wc-sets-list");
      session.loggedSets.forEach((s, idx) => {
        const meta = getExerciseMeta(s.name, this.plugin.programStore);
        const wText = s.weight > 0 ? ` +${s.weight} кг` : "";
        const isPR = this.lastRecordSetLineIndex === s.lineIndex;
        const pill = pillsList.createDiv(
          "my-wc-set-pill" + (isPR ? " is-record" : "")
        );
        pill.createSpan({
          text: `${idx + 1}. ${s.name}: ${s.reps} ${meta.unit}${wText}${isPR ? " 🏆" : ""}`,
        });

function sanitizeCatalog(raw: unknown): ExerciseCatalogItem[] {
  const defaults = cloneDefaultCatalog();
  if (!Array.isArray(raw) || raw.length === 0) return defaults;

  const out: ExerciseCatalogItem[] = [];
  const seen = new Set<string>();
  const defaultMap = new Map(
    defaults.map((d) => [d.name.toLowerCase(), d])
  );

  for (const item of raw) {
    if (!item || typeof item !== "object") continue;
    const c = item as Partial<ExerciseCatalogItem>;
    const name = typeof c.name === "string" ? c.name.trim() : "";
    if (!name) continue;
    const low = name.toLowerCase();
    if (seen.has(low)) continue;
    seen.add(low);

    const defMatch = defaultMap.get(low);
    const isSec = c.field === "секунды" || c.unit === "с";
    out.push({
      name,
      field: isSec ? "секунды" : "повторы",
      unit: isSec ? "с" : "повт.",
      defaultWeight:
        typeof c.defaultWeight === "number"
          ? c.defaultWeight
          : defMatch?.defaultWeight ?? 0,
      defaultSets:
        typeof c.defaultSets === "number"
          ? c.defaultSets
          : defMatch?.defaultSets ?? 4,
      defaultTarget:
        typeof c.defaultTarget === "string" && c.defaultTarget.trim()
          ? c.defaultTarget.trim()
          : defMatch?.defaultTarget ?? "MAX",
      defaultRest:
        typeof c.defaultRest === "number"
          ? c.defaultRest
          : defMatch?.defaultRest ?? 120,
      isQuickBase:
        typeof c.isQuickBase === "boolean"
          ? c.isQuickBase
          : Boolean(defMatch?.isQuickBase || c.isCustom),
      isCustom: Boolean(c.isCustom),
    });
  }

  for (const def of defaults) {
    if (!seen.has(def.name.toLowerCase())) {
      seen.add(def.name.toLowerCase());
      out.push({ ...def });
    }
  }

  return out;
}

export class ProgramStore {
  private app: App;
  private filePath: string;
  private data: ProgramStoreData;
  private saveTimer: number | null = null;

  constructor(app: App, pluginDir: string) {
    this.app = app;
    this.filePath = normalizePath(`${pluginDir}/program.json`);
    this.data = {
      version: 3,
      catalog: cloneDefaultCatalog(),
    };
  }

  async load(legacySettingsRaw?: unknown): Promise<void> {
    const adapter = this.app.vault.adapter;
    let loadedFromDisk = false;

    try {
      if (await adapter.exists(this.filePath)) {
        const rawText = await adapter.read(this.filePath);
        const parsed = JSON.parse(rawText) as Record<string, unknown>;
        if (parsed && typeof parsed === "object") {
          this.data = {
            version: 3,
            catalog: sanitizeCatalog(parsed.catalog),
          };
          loadedFromDisk = true;
        }
      }
    } catch {
      loadedFromDisk = false;
    }

    if (
      !loadedFromDisk &&
      legacySettingsRaw &&
      typeof legacySettingsRaw === "object"
    ) {
      const leg = legacySettingsRaw as Record<string, unknown>;
      this.data = {
        version: 3,
        catalog: sanitizeCatalog(leg.catalog),
      };
      await this.saveNow();
    } else if (!loadedFromDisk) {
      await this.saveNow();
    }
  }

  getCatalog(): readonly ExerciseCatalogItem[] {
    return this.data.catalog;
  }

  getQuickExercises(): readonly ExerciseCatalogItem[] {
    return this.data.catalog.filter((c) => c.isQuickBase || c.isCustom);
  }

  findExercise(name: string): ExerciseCatalogItem | undefined {
    const low = name.trim().toLowerCase();
    return this.data.catalog.find((c) => c.name.toLowerCase() === low);
  }

  addCustomExercise(
    name: string,
    field: "повторы" | "секунды"
  ): ExerciseCatalogItem | null {
    const clean = name.replace(/\s+/g, " ").trim();
    if (!clean) return null;
    const existing = this.findExercise(clean);
    if (existing) {
      existing.isQuickBase = true;
      this.requestSave();
      return existing;
    }

    const item: ExerciseCatalogItem = {
      name: clean,
      field,
      unit: field === "секунды" ? "с" : "повт.",
      defaultWeight: 0,
      defaultSets: 4,
      defaultTarget: "MAX",
      defaultRest: 120,
      isQuickBase: true,
      isCustom: true,
    };
    this.data.catalog.push(item);
    this.requestSave();
    return item;
  }

  toggleQuickButton(name: string, enabled: boolean): void {
    const item = this.findExercise(name);
    if (item) {
      item.isQuickBase = enabled;
      this.requestSave();
    }
  }

  removeExerciseFromCatalog(name: string): boolean {
    const low = name.trim().toLowerCase();
    const idx = this.data.catalog.findIndex(
      (c) => c.name.toLowerCase() === low
    );
    if (idx === -1) return false;
    this.data.catalog.splice(idx, 1);
    this.requestSave();
    return true;
  }

  resetCatalogToDefaults(): void {
    this.data.catalog = cloneDefaultCatalog();
    this.requestSave();
  }

  requestSave(): void {
    if (this.saveTimer !== null) window.clearTimeout(this.saveTimer);
    this.saveTimer = window.setTimeout(() => {
      this.saveTimer = null;
      void this.saveNow();
    }, 200);
  }

  async saveNow(): Promise<void> {
    try {
      await this.app.vault.adapter.write(
        this.filePath,
        JSON.stringify(this.data, null, 2)
      );
    } catch {
      // игнорируем ошибки адаптера
    }
  }

  flush(): void {
    if (this.saveTimer !== null) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
      void this.saveNow();
    }
  }
}

export function todayLocalISO(): string {

export function currentTimeHHMM(): string {
  const d = new Date();
  const p = (n: number) => String(n).padStart(2, "0");

}

export function isoWeekdayOf(iso: string): number {
  const parts = iso.split("-").map(Number);
  const y = parts[0];
  const m = parts[1];
  const d = parts[2];
  if (!y || !m || !d) {
    const day = new Date().getDay();
    return day === 0 ? 7 : day;
  }
  const dt = new Date(y, m - 1, d);
  const day = dt.getDay();
  return day === 0 ? 7 : day;
}

export function formatShortDateRu(iso: string): string {
  const parts = iso.split("-");
  if (parts.length !== 3) return iso;
  return `${parts[2]}.${parts[1]}`;
}

export function formatDateWithWeekdayRu(iso: string): string {
  const wNum = isoWeekdayOf(iso);
  const wFull = DAYS_FULL[wNum - 1] ?? "Сегодня";
  const parts = iso.split("-");
  if (parts.length === 3) {
    return `${wFull}, ${parts[2]}.${parts[1]}.${parts[0]}`;
  }
  return wFull;
}

export function formatDefaultSessionTitle(

): string {
  const wNum = isoWeekdayOf(dateISO);
  const dayFull = DAYS_FULL[wNum - 1] ?? "Тренировка";

    if (!uniqueNames.includes(p.name)) uniqueNames.push(p.name);
  }
  if (uniqueNames.length === 0) {

function readStr(v: unknown): string {
  return v === null || v === undefined ? "" : String(v).trim();
}

function readNum(v: unknown): number | null {
  if (typeof v === "number") return Number.isFinite(v) ? v : null;
  if (typeof v !== "string") return null;
  const s = v.trim().replace(",", ".");
  if (!s) return null;
  const n = Number(s);
  return Number.isFinite(n) ? n : null;
}

function extractISODate(v: unknown): string {
  const s = readStr(v).slice(0, 10);
  return /^\d{4}-\d{2}-\d{2}$/.test(s) ? s : "";
}

export function readOptionalProgramDay(
  fm: Record<string, unknown>
): number | null {
  const explicit = readNum(fm[WORKOUT_KEYS.programDay]);
  if (explicit && explicit >= 1 && explicit <= 7) {
    return Math.round(explicit);
  }
  return null;
}

export function parseSetsFromContent(content: string): LoggedWorkoutSet[] {
  const sets: LoggedWorkoutSet[] = [];
  if (!content) return sets;
  const lines = content.split(/\r?\n/);

  lines.forEach((line, idx) => {
    const m = line.match(SET_LINE_REGEX);
    if (!m) return;
    const name = (m[1] ?? "").trim();
    const fieldRaw = (m[2] ?? "").trim();
    const field: ExerciseFieldKind =
      fieldRaw === "секунды" ? "секунды" : "повторы";
    const val = Number((m[3] ?? "").replace(",", "."));
    const weight = Number((m[4] ?? "").replace(",", "."));
    const variant = (m[5] ?? "").trim() || "Основной";
    if (
      name &&
      Number.isFinite(val) &&
      val > 0 &&
      Number.isFinite(weight) &&
      weight >= 0
    ) {
      sets.push({ lineIndex: idx, name, field, reps: val, weight, variant });
    }
  });

  return sets;
}

export function collectWorkoutSessions(
  app: App,
  settings: WorkoutPluginSettings
): WorkoutSessionSummary[] {
  const folderPrefix =
    normalizePath(settings.workoutsFolder || "Знания").replace(/\/+$/, "") + "/";
  const result: WorkoutSessionSummary[] = [];

  for (const file of app.vault.getMarkdownFiles()) {
    if (!file.path.startsWith(folderPrefix)) continue;
    const fm = app.metadataCache.getFileCache(file)?.frontmatter;
    const isWorkoutByFm =
      fm && readStr(fm[WORKOUT_KEYS.type]) === WORKOUT_TYPE_VALUE;
    const isWorkoutByName = /^Тренировка\s+\d{4}-\d{2}-\d{2}$/i.test(
      file.basename
    );
    if (!isWorkoutByFm && !isWorkoutByName) continue;

    const fallbackDate = extractISODate(
      file.basename.replace(/^Тренировка\s+/i, "")
    );
    const dateISO =
      (fm ? extractISODate(fm[WORKOUT_KEYS.date]) : "") || fallbackDate;
    const programDay = fm ? readOptionalProgramDay(fm) : null;
    const durationMin = fm ? readNum(fm[WORKOUT_KEYS.durationMin]) : null;
    const startStr = fm ? readStr(fm[WORKOUT_KEYS.start]) : "";
    const title = (fm ? readStr(fm[WORKOUT_KEYS.title]) : "") || file.basename;

    result.push({
      file,
      path: file.path,
      dateISO,
      programDay,
      durationMin,
      startStr,
      title,
    });
  }

  result.sort(
    (a, b) =>
      b.dateISO.localeCompare(a.dateISO) || b.path.localeCompare(a.path, "ru")
  );
  return result;
}

export interface PreviousExerciseReference {
  dateISO: string;
  sets: number[];
  best: number;
  lastReps: number;
  weight: number;
  allTimeBestForExactWeight: number;
}

export async function findPreviousExerciseResult(
  app: App,
  allSessions: WorkoutSessionSummary[],
  currentSessionPath: string,
  exerciseName: string,
  targetWeight: number
): Promise<PreviousExerciseReference | null> {
  const candidates = allSessions.filter((s) => s.path !== currentSessionPath);

  let latestExact: { dateISO: string; sets: number[]; best: number } | null =
    null;
  let fallbackOtherWeight: {
    dateISO: string;
    sets: number[];
    best: number;
    weight: number;
  } | null = null;
  let allTimeBestExact = 0;

  for (const sess of candidates) {
    let content = "";
    try {
      content = await app.vault.cachedRead(sess.file);
    } catch {
      continue;
    }
    const sets = parseSetsFromContent(content).filter(
      (s) => s.name === exerciseName
    );
    if (sets.length === 0) continue;

    const exactSets = sets.filter(
      (s) => Math.abs(s.weight - targetWeight) < 0.01
    );
    if (exactSets.length > 0) {
      const vals = exactSets.map((s) => s.reps);
      const sessBest = Math.max(...vals);
      if (sessBest > allTimeBestExact) {
        allTimeBestExact = sessBest;
      }
      if (!latestExact) {
        latestExact = { dateISO: sess.dateISO, sets: vals, best: sessBest };
      }
    } else if (!fallbackOtherWeight) {
      const firstW = sets[0]?.weight ?? 0;
      const sameW = sets
        .filter((s) => Math.abs(s.weight - firstW) < 0.01)
        .map((s) => s.reps);
      fallbackOtherWeight = {
        dateISO: sess.dateISO,
        sets: sameW,
        best: Math.max(...sameW),
        weight: firstW,
      };
    }
  }

  if (latestExact) {
    return {
      dateISO: latestExact.dateISO,
      sets: latestExact.sets,
      best: latestExact.best,
      lastReps: latestExact.sets[0] ?? 10,
      weight: targetWeight,
      allTimeBestForExactWeight: allTimeBestExact,
    };
  }

  if (fallbackOtherWeight) {
    return {
      dateISO: fallbackOtherWeight.dateISO,
      sets: fallbackOtherWeight.sets,
      best: fallbackOtherWeight.best,
      lastReps: fallbackOtherWeight.sets[0] ?? 10,
      weight: fallbackOtherWeight.weight,
      allTimeBestForExactWeight: 0,
    };
  }

  return null;
}

export async function computeAllExerciseRecords(
  app: App,
  allSessions: WorkoutSessionSummary[],
  store?: ProgramStore
): Promise<ExerciseRecordEntry[]> {
  interface SessionGroup {
    dateISO: string;
    path: string;
    best: number;
  }

  const byKey = new Map<
    string,
    {
      name: string;
      variant: string;
      weight: number;
      unit: "повт." | "с";
      history: SessionGroup[];
    }
  >();

  for (const sess of allSessions) {
    let content = "";
    try {
      content = await app.vault.cachedRead(sess.file);
    } catch {
      continue;
    }
    const sets = parseSetsFromContent(content);
    if (sets.length === 0) continue;

    const perSessionMax = new Map<
      string,
      { name: string; variant: string; weight: number; best: number }
    >();
    for (const s of sets) {
      const key = JSON.stringify([s.name, s.variant, s.weight]);
      const prev = perSessionMax.get(key);
      if (!prev || s.reps > prev.best) {
        perSessionMax.set(key, {
          name: s.name,
          variant: s.variant,
          weight: s.weight,
          best: s.reps,
        });
      }
    }

    for (const [key, g] of perSessionMax.entries()) {
      let entry = byKey.get(key);
      if (!entry) {
        const meta = getExerciseMeta(g.name, store);
        entry = {
          name: g.name,
          variant: g.variant,
          weight: g.weight,
          unit: meta.unit,
          history: [],
        };
        byKey.set(key, entry);
      }
      entry.history.push({
        dateISO: sess.dateISO,
        path: sess.path,
        best: g.best,
      });
    }
  }

  const out: ExerciseRecordEntry[] = [];
  for (const [key, val] of byKey.entries()) {
    const latest = val.history[0];
    if (!latest) continue;
    const previous = val.history[1] ?? null;
    const allTimeBest = Math.max(...val.history.map((h) => h.best));
    out.push({
      key,
      name: val.name,
      variant: val.variant,
      weight: val.weight,
      unit: val.unit,
      allTimeBest,
      latestBest: latest.best,
      previousBest: previous ? previous.best : null,
      latestDateISO: latest.dateISO,
      latestSessionPath: latest.path,
    });
  }

  return out;
}

        await app.vault.createFolder(current);
      } catch {
        // уже существует
      }
    }
  }
}

export function clearManualHabitOverrideForDate(

export async function syncWorkoutHabitToDiary(
  app: App,
  settings: WorkoutPluginSettings,
  targetDateISO: string
): Promise<void> {
  if (!targetDateISO) return;
  clearManualHabitOverrideForDate(targetDateISO, WORKOUT_KEYS.diaryWorkoutFlag);

  const diaryFolder = normalizePath(settings.diaryFolder || "Ежедневник").replace(
    /\/+$/,
    ""
  );
  const directPath = normalizePath(`${diaryFolder}/${targetDateISO}.md`);
  let diaryFile: TFile | null = null;

  const directAbstract = app.vault.getAbstractFileByPath(directPath);
  if (directAbstract instanceof TFile) {
    diaryFile = directAbstract;
  } else {
    const prefix = `${diaryFolder}/`;
    for (const f of app.vault.getMarkdownFiles()) {
      if (!f.path.startsWith(prefix)) continue;
      const fm = app.metadataCache.getFileCache(f)?.frontmatter;
      if (fm && extractISODate(fm[WORKOUT_KEYS.date]) === targetDateISO) {
        diaryFile = f;
        break;
      }
    }
  }

  if (!diaryFile) {
    await ensureFolder(app, diaryFolder);
    const content = [
      "---",
      `${WORKOUT_KEYS.date}: "${targetDateISO}"`,
      `${WORKOUT_KEYS.diaryWorkoutFlag}: true`,
      "---",
      "",
    ].join("\n");
    await app.vault.create(directPath, content);
    return;
  }

  await app.fileManager.processFrontMatter(
    diaryFile,
    (fm: Record<string, unknown>) => {
      fm[WORKOUT_KEYS.diaryWorkoutFlag] = true;
    }
  );
}

export function computeSessionDurationMinutes(
  startStr: string,
  fallbackMin: number | null,
  sessionDateISO?: string
): number {
  if (/^\d{2}:\d{2}$/.test(startStr)) {
    const hh = Number(startStr.slice(0, 2));
    const mm = Number(startStr.slice(3, 5));
    if (
      Number.isInteger(hh) &&
      hh >= 0 &&
      hh <= 23 &&
      Number.isInteger(mm) &&
      mm >= 0 &&
      mm <= 59
    ) {
      const now = new Date();
      if (sessionDateISO && /^\d{4}-\d{2}-\d{2}$/.test(sessionDateISO)) {
        const parts = sessionDateISO.split("-").map(Number);
        const y = parts[0];
        const m = parts[1];
        const d = parts[2];
        if (y && m && d) {
          const startDt = new Date(y, m - 1, d, hh, mm, 0, 0);
          let diffMin = Math.round((now.getTime() - startDt.getTime()) / 60000);
          if (diffMin < 0) diffMin += 1440;
          if (diffMin > 0 && diffMin <= 600) return diffMin;
        }
      }
      const nowTotal = now.getHours() * 60 + now.getMinutes();
      const startTotal = hh * 60 + mm;
      let diff = nowTotal - startTotal;
      if (diff < 0) diff += 1440;
      if (diff > 0 && diff <= 600) return diff;
    }
  }
  return fallbackMin && fallbackMin > 0 ? fallbackMin : 60;
}

import { Notice } from "obsidian";
import type MyWorkoutCompanionPlugin from "./main";
import type { ActiveRestTimerState } from "./types";

export type TimerTickListener = (state: ActiveRestTimerState, now: number) => void;

export class WorkoutRestTimerService {
  private plugin: MyWorkoutCompanionPlugin;
  private intervalId: number | null = null;
  private listeners = new Set<TimerTickListener>();
  private audioCtx: AudioContext | null = null;
  private lastFinishedBannerEndAt = 0;

  constructor(plugin: MyWorkoutCompanionPlugin) {
    this.plugin = plugin;
  }

  /**
   * Вызывается внутри обработчика клика пользователя (старт таймера или запись подхода),
   * чтобы разблокировать AudioContext на iOS / iPadOS.
   */
  unlockAudioOnUserGesture(): void {
    if (!this.plugin.settings.soundEnabled) return;
    try {
      const Ctx =
        window.AudioContext ||
        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
      if (!Ctx) return;
      if (!this.audioCtx) {
        this.audioCtx = new Ctx();
      }
      if (this.audioCtx.state === "suspended") {
        void this.audioCtx.resume();
      }
    } catch {
      // игнорируем, если аудио недоступно на устройстве
    }
  }

  private playCompletionChime(): void {
    if (!this.plugin.settings.soundEnabled) return;
    try {
      const Ctx =
        window.AudioContext ||
        (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
      if (!Ctx) return;
      if (!this.audioCtx) {
        this.audioCtx = new Ctx();
      }
      const ctx = this.audioCtx;
      if (ctx.state === "suspended") {
        void ctx.resume();
      }

      const now = ctx.currentTime;
      const playTone = (freq: number, startOffset: number, duration: number) => {
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(freq, now + startOffset);
        gain.gain.setValueAtTime(0.0001, now + startOffset);
        gain.gain.exponentialRampToValueAtTime(0.18, now + startOffset + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + startOffset + duration);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now + startOffset);
        osc.stop(now + startOffset + duration + 0.02);
      };

      playTone(659.25, 0, 0.16); // E5
      playTone(880.0, 0.18, 0.24); // A5
    } catch {
      // игнорируем ошибки аудио на заблокированном устройстве
    }
  }

  getState(): ActiveRestTimerState {
    return this.plugin.settings.timer;
  }

  getLastFinishedBannerEndAt(): number {
    return this.lastFinishedBannerEndAt;
  }

  start(sessionPath: string, durationSec: number): void {
    this.unlockAudioOnUserGesture();
    const sec = Math.max(5, Math.round(durationSec));
    const now = Date.now();
    const endAt = now + sec * 1000;
    this.lastFinishedBannerEndAt = 0;

    this.plugin.settings.timer = {
      sessionPath,
      startAt: now,
      endAt,
      durationSec: sec,
      notifiedEndAt: 0,
    };
    this.plugin.requestSaveSettings();
    this.ensureLoop();
    this.emitTick();
  }

  addSeconds(deltaSec: number): void {
    this.unlockAudioOnUserGesture();
    const t = this.plugin.settings.timer;
    const now = Date.now();
    if (t.endAt > now) {
      const nextEnd = Math.max(now + 1000, t.endAt + deltaSec * 1000);
      t.endAt = nextEnd;
      t.durationSec = Math.max(5, Math.round((nextEnd - t.startAt) / 1000));
      this.plugin.requestSaveSettings();
      this.emitTick();
    } else {
      const base = Math.max(15, deltaSec > 0 ? deltaSec : t.durationSec || 120);
      this.start(t.sessionPath, base);
    }
  }

  reset(newDefaultDurationSec?: number): void {
    const t = this.plugin.settings.timer;
    this.lastFinishedBannerEndAt = 0;
    this.plugin.settings.timer = {
      sessionPath: t.sessionPath,
      startAt: 0,
      endAt: 0,
      durationSec: newDefaultDurationSec ?? (t.durationSec || 120),
      notifiedEndAt: t.notifiedEndAt,
    };
    this.plugin.requestSaveSettings();
    this.stopLoop();
    this.emitTick();
  }

  subscribe(fn: TimerTickListener): () => void {
    this.listeners.add(fn);
    this.checkAndSyncNow();
    return () => {
      this.listeners.delete(fn);
    };
  }

  /**
   * Вызывается при старте плагина, открытии окна, а также по событиям visibilitychange / focus на iPad.
   * Не затирает молча истёкший таймер, а фиксирует завершение и уведомляет пользователя.
   */
  checkAndSyncNow(): void {
    const t = this.plugin.settings.timer;
    const now = Date.now();

    if (t.endAt > 0) {
      if (t.endAt > now) {
        this.ensureLoop();
      } else {
        const finishedEndAt = t.endAt;
        const wasNotified = t.notifiedEndAt === finishedEndAt;
        this.lastFinishedBannerEndAt = finishedEndAt;
        t.endAt = 0;
        t.startAt = 0;
        t.notifiedEndAt = finishedEndAt;
        this.plugin.requestSaveSettings();
        this.stopLoop();

        if (!wasNotified) {
          const overdueSec = Math.max(0, Math.round((now - finishedEndAt) / 1000));
          const suffix =
            overdueSec >= 5 ? ` (${overdueSec} с назад)` : "";
          new Notice(`⏰ Отдых завершён${suffix} — пора на подход!`);
          this.playCompletionChime();
        }
      }
    }
    this.emitTick();
  }

  private ensureLoop(): void {
    if (this.intervalId !== null) return;
    this.intervalId = window.setInterval(() => {
      this.checkAndSyncNow();
    }, 250);
  }

  private stopLoop(): void {
    if (this.intervalId !== null) {
      window.clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }

  private emitTick(): void {
    const now = Date.now();
    const state = this.plugin.settings.timer;
    for (const fn of this.listeners) {
      fn(state, now);
    }
  }

  destroy(): void {
    this.stopLoop();
    this.listeners.clear();
    if (this.audioCtx) {
      try {
        void this.audioCtx.close();
      } catch {
        // игнорируем
      }
      this.audioCtx = null;
    }
  }
}

export function formatSecondsMMSS(sec: number): string {
  const safe = Math.max(0, Math.floor(sec));
  const m = String(Math.floor(safe / 60)).padStart(2, "0");
  const s = String(safe % 60).padStart(2, "0");
  return `${m}:${s}`;
}
